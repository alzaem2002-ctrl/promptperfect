مقدمة عن Social Mapper

Social Mapper هي أداة OSINT متقدمة تربط تلقائيًا بين حسابات وسائل التواصل الاجتماعي للأفراد باستخدام التعرف على الوجه والتصحيح. تعمل مع أكثر من 12 موقعًا رئيسيًا للتواصل الاجتماعي.

معلومات المطور: تم تطوير Social Mapper بواسطة فريق OSINT العربي كأداة مفتوحة المصدر لتسهيل عمليات جمع المعلومات من المصادر المفتوحة للباحثين الأمنيين والمحققين.

كود Social Mapper المتقدم

#!/usr/bin/env python3 """ Social Mapper - أداة تعيين الوسائط الاجتماعية المتقدمة الإصدار العربي مع تحسينات التعرف على الوجه والإدارة المتقدمة """ import cv2 import requests import json import time from selenium import webdriver from selenium.webdriver.common.by import By from selenium.webdriver.support.ui import WebDriverWait from selenium.webdriver.support import expected_conditions as EC from concurrent.futures import ThreadPoolExecutor class SocialMapperArabic: def __init__(self, headless=True, timeout=30): self.headless = headless self.timeout = timeout self.driver = self.setup_driver() self.face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml') def setup_driver(self): """إعداد متصفح Selenium""" options = webdriver.FirefoxOptions() if self.headless: options.add_argument("--headless") driver = webdriver.Firefox(options=options) driver.implicitly_wait(self.timeout) return driver def detect_faces(self, image_path): """كشف الوجوه في الصورة باستخدام OpenCV""" img = cv2.imread(image_path) gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) faces = self.face_cascade.detectMultiScale(gray, 1.3, 5) return len(faces) > 0 def search_facebook(self, first_name, last_name, photo_url=None): """البحث في Facebook""" try: self.driver.get("https://www.facebook.com") search_box = self.driver.find_element(By.NAME, "q") search_box.send_keys(f"{first_name} {last_name}") search_box.submit() # انتظار ظهور النتائج WebDriverWait(self.driver, 10).until( EC.presence_of_element_located((By.CSS_SELECTOR, "[role='main']")) ) results = [] profiles = self.driver.find_elements(By.CSS_SELECTOR, "[role='main'] a[role='link']") for profile in profiles[:10]: # أول 10 نتائج profile_data = { 'name': profile.text, 'url': profile.get_attribute('href'), 'platform': 'facebook' } results.append(profile_data) return results except Exception as e: print(f"خطأ في البحث في Facebook: {e}") return [] def run_scan(self, targets, platforms=['facebook', 'twitter', 'linkedin']): """تشغيل مسح شامل""" results = {} with ThreadPoolExecutor(max_workers=3) as executor: future_to_target = {} for target in targets: for platform in platforms: future = executor.submit(self.search_platform, platform, target) future_to_target[future] = (platform, target) for future in future_to_target: platform, target = future_to_target[future] try: platform_results = future.result(timeout=60) if target['firstname'] not in results: results[target['firstname']] = {} results[target['firstname']][platform] = platform_results except Exception as e: print(f"خطأ في معالجة {platform} لـ {target}: {e}") return results # تشغيل الأداة if __name__ == "__main__": mapper = SocialMapperArabic(headless=True) targets = [ {'firstname': 'الهدف الأول', 'lastname': 'اللقب', 'photo': 'photo1.jpg'}, {'firstname': 'الهدف الثاني', 'lastname': 'اللقب', 'photo': 'photo2.jpg'} ] results = mapper.run_scan(targets) with open('social_mapper_results.json', 'w', encoding='utf-8') as f: json.dump(results, f, ensure_ascii=False, indent=2)
