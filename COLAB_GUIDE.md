## ?? ???? ????? (All-in-One)

??? ???? ??? ???? ???? ?? ???:

```python
#!/usr/bin/env python3
"""
ComfyUI + AnimateDiff Setup for Google Colab
All-in-One Script
"""

import os
import subprocess
import time
import re

print("?? Starting ComfyUI Setup on Google Colab...")
print("=" * 60)

# 1. Check GPU
print("\n[1/10] Checking GPU...")
os.system('nvidia-smi')

# 2. Clone ComfyUI
print("\n[2/10] Cloning ComfyUI...")
os.chdir('/content')
os.system('git clone https://github.com/comfyanonymous/ComfyUI')

# 3. Install dependencies
print("\n[3/10] Installing dependencies...")
os.chdir('/content/ComfyUI')
os.system('pip install -q -r requirements.txt')
os.system('pip install -q xformers einops')

# 4. Create directories
print("\n[4/10] Creating model directories...")
dirs = [
    "/content/ComfyUI/models/checkpoints",
    "/content/ComfyUI/models/vae",
    "/content/ComfyUI/models/animatediff_models"
]
for d in dirs:
    os.makedirs(d, exist_ok=True)

# 5-7. Download models
print("\n[5/10] Downloading Stable Diffusion v1.5...")
os.system('wget -q -P /content/ComfyUI/models/checkpoints https://huggingface.co/runwayml/stable-diffusion-v1-5/resolve/main/v1-5-pruned-emaonly.safetensors')

print("\n[6/10] Downloading VAE...")
os.system('wget -q -P /content/ComfyUI/models/vae https://huggingface.co/stabilityai/sd-vae-ft-mse-original/resolve/main/vae-ft-mse-840000-ema-pruned.safetensors')

print("\n[7/10] Downloading AnimateDiff...")
os.system('wget -q -P /content/ComfyUI/models/animatediff_models https://huggingface.co/guoyww/animatediff/resolve/main/mm_sd_v15_v2.ckpt')

# 8. Install cloudflared
print("\n[8/10] Installing Cloudflared...")
os.system('wget -q https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64.deb')
os.system('dpkg -i cloudflared-linux-amd64.deb > /dev/null 2>&1')

# 9. Start ComfyUI
print("\n[9/10] Starting ComfyUI Server...")
comfy_proc = subprocess.Popen(
    ['python', 'main.py', '--listen', '0.0.0.0'],
    stdout=subprocess.PIPE,
    stderr=subprocess.PIPE
)
time.sleep(15)

# 10. Create Cloudflare Tunnel
print("\n[10/10] Creating Cloudflare Tunnel...")
tunnel_proc = subprocess.Popen(
    ['cloudflared', 'tunnel', '--url', 'http://localhost:8188'],
    stdout=subprocess.PIPE,
    stderr=subprocess.PIPE,
    universal_newlines=True
)

time.sleep(10)

# Extract URL
tunnel_url = None
for i in range(30):
    line = tunnel_proc.stderr.readline()
    if 'trycloudflare.com' in line:
        match = re.search(r'https://[a-zA-Z0-9-]+\.trycloudflare\.com', line)
        if match:
            tunnel_url = match.group(0)
            break
    time.sleep(0.5)

# Display results
print("\n" + "=" * 60)
if tunnel_url:
    print("? SUCCESS! ComfyUI is ready!")
    print("=" * 60)
    print(f"\n?? Public URL: {tunnel_url}")
    print(f"\n?? iPad Setup (Comfy Remote):")
    print(f"   Host: {tunnel_url.replace('https://', '')}")
    print(f"   Port: 443")
    print(f"   HTTPS: ?")
    print("\n?? Keep Colab running!")
else:
    print("?? Tunnel created but URL not extracted")
    print("Check stderr output above")
print("=" * 60)
```
